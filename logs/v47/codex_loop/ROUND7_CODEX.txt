from sklearn.metrics import roc_auc_score
from sklearn.model_selection import StratifiedKFold

heading = "## Round7 corrected digit LightGBM recipe"

    with report.open("a") as f:
        f.write(

            "Round7 used a different feature recipe. Its numeric-only input omitted the native "
            "categorical columns and five categorical cents features used in the earlier "
            "0.9368051080303192 solo run. The original Round7 result remains NO-GO "
            "(−0.848928e-5, 2/5). This rerun restores the earlier recipe and scores Fold-0 "
            "only. NO_SUBMIT=True, SUBMIT_ARMED=False, RunPod $0, tip CSVs read 0.\n"
        )

pred = Path("preds/v47_sovereign")
df = pd.read_csv("data/train.csv")
y = df.pop("is_fraud").to_numpy()
x = df.drop(columns="id")
for col in x.select_dtypes(include=["object", "str"]).columns:
    x[col] = x[col].astype("category")
for col in [
    "amount_usd", "account_balance_usd", "distance_from_home_km",
    "merchant_risk_score", "velocity_score",
]:
    x[col + "_cents"] = (np.rint(100 * x[col]).astype("int64") % 100).astype("category")

folds = np.load(pred / "folds_skf42.npy")
train, valid = next(StratifiedKFold(5, shuffle=True, random_state=42).split(x, y))
np.testing.assert_array_equal(valid, np.flatnonzero(folds == 0))
assert len(valid) == 139992
y0 = y[valid]
base = np.load(pred / "z_sovereign_v0_oof.npy").ravel()[valid]
base_auc = roc_auc_score(y0, base)
assert abs(base_auc - 0.9512507991341679) < 1e-12

model = LGBMClassifier(
    n_estimators=450, learning_rate=0.045, num_leaves=31,
    min_child_samples=120, reg_lambda=5, n_jobs=4,
    verbosity=-1, random_state=42,
)
model.fit(x.iloc[train], y[train])
digit = model.predict(x.iloc[valid], raw_score=True)
solo_auc = roc_auc_score(y0, digit)
print(f"Corrected digit solo AUC {solo_auc:.16f}", flush=True)
if abs(solo_auc - 0.9368051080303192) > 1e-6:
    raise SystemExit("Earlier solo AUC did not reproduce. Diagnose recipe or runtime drift before choosing weights.")

def rank_logit(scores):
    return logit((rankdata(scores) - 0.5) / len(scores))

anchor = rank_logit(base)
residual = rank_logit(digit) - anchor
grid = (0.0, -0.025, 0.025, -0.05, 0.05, -0.075, 0.075, -0.10, 0.10, -0.125, 0.125, -0.15, 0.15)
candidate = np.full(len(y0), np.nan)
weights, sub = [], []
for fit, score in StratifiedKFold(5, shuffle=True, random_state=7).split(anchor, y0):
    weight = max(grid, key=lambda w: roc_auc_score(y0[fit], anchor[fit] + w * residual[fit]))
    candidate[score] = anchor[score] + weight * residual[score]
    weights.append(weight)
    sub.append((roc_auc_score(y0[score], candidate[score]) - roc_auc_score(y0[score], base[score])) * 1e5)

assert np.isfinite(candidate).all()
cand_auc = roc_auc_score(y0, candidate)
delta = cand_auc - base_auc
pos = sum(d > 0 for d in sub)
verdict = "DEV_GATE_PASS" if delta >= 8e-5 and pos >= 4 else "NO-GO"
result = {
    "scope": "F0_DEV_ONLY",
    "base_auc": base_auc, "solo_auc": solo_auc, "cand_auc": cand_auc,
    "delta_e5": delta * 1e5, "pos5": pos, "sub_e5": sub,
    "weights": weights, "grid": grid, "verdict": verdict,
    "milestone": "LOCAL_GATE_PASS", "NO_SUBMIT": True, "SUBMIT_ARMED": False,
}
payload = json.dumps(result, indent=2)
Path("logs/v47/codex_loop/ROUND7_F0_SIGNED_LGBM_CORRECTED.json").write_text(payload + "\n")
np.save(pred / "lgbm_digit_corrected_solo_f0.npy", digit)
np.save(pred / "lgbm_digit_corrected_signed_f0.npy", candidate)
with report.open("a") as f:
    f.write(
        f"\nCorrected Fold-0 dev result — base {base_auc:.16f}, solo {solo_auc:.16f}, "
        f"candidate {cand_auc:.16f}, Δe-5 {delta * 1e5:+.12f}, {pos}/5, {verdict}. "
        f"Weights {weights}. Subfold Δe-5 {sub}. Milestone remains LOCAL_GATE_PASS.\n"
    )
print(payload)
