def pair_rate_features(fit_frame, fit_labels, score_frame):

    ]
    pairs = categorical_pairs + numeric_pairs

    def encode(reference, observed, query):
        prior = observed.mean()
        edges = {
            column: np.unique(reference[column].quantile(np.arange(1, 16) / 16).to_numpy())
            for column in sorted({column for pair in numeric_pairs for column in pair})
        }

        def keys(features, left, right):
            def values(column):
                if column in edges:
                    return pd.Series(
                        np.searchsorted(edges[column], features[column].to_numpy(), side="right"),
                        index=features.index,
                    ).astype(str)
                return features[column].astype("string").fillna("__missing__")
            return values(left).str.cat(values(right), sep="|")

        columns = []
        for left, right in pairs:
            reference_key = keys(reference, left, right)
            table = pd.DataFrame(
                {"key": reference_key.to_numpy(), "label": observed}
            ).groupby("key")["label"].agg(["sum", "size"])
            rate = (table["sum"] + 100 * prior) / (table["size"] + 100)
            columns.append(keys(query, left, right).map(rate).fillna(prior).to_numpy(dtype=float))
        return np.column_stack(columns)

    train_rates = np.full((len(fit_frame), len(pairs)), np.nan)
    for ref, held in StratifiedKFold(3, shuffle=True, random_state=2028).split(fit_frame, fit_labels):
        train_rates[held] = encode(fit_frame.iloc[ref], fit_labels[ref], fit_frame.iloc[held])
    score_rates = encode(fit_frame, fit_labels, score_frame)
    assert np.isfinite(train_rates).all() and np.isfinite(score_rates).all()
    names = [f"pair_rate__{left}__{right}" for left, right in pairs]
    return (
        fit_frame.assign(**dict(zip(names, train_rates.T))),
        score_frame.assign(**dict(zip(names, score_rates.T))),
    )
""")
